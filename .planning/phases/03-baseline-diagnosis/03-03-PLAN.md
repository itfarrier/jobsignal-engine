    ```javascript
    // Parse n8n execution runData for baseline metrics
    // Logic matches verified parse-runData.mjs from Plan 02
    const input = $input.first().json;
    const executionData = input.results?.[0] || input.data?.[0] || input;
    const runData = executionData.data?.runData || {};

    function sumRealEntries(nodeName) {
      const entries = runData[nodeName] || [];
      return entries.reduce((total, iter) => {
        const items = iter.output?.[0] || [];
        const realItems = items.filter(i => !i.json?._empty);
        return total + realItems.length;
      }, 0);
    }

    function sumAllEntries(nodeName) {
      const entries = runData[nodeName] || [];
      return entries.reduce((total, iter) => {
        return total + (iter.output?.[0]?.length || 0);
      }, 0);
    }

    const feedsGenerated = runData['Build Feed List']?.[0]?.output?.[0]?.length || 0;
    const totalRssItems = sumRealEntries('RSS Feed Read');
    const afterGeoFilter = sumRealEntries('Parse & Filter Jobs');
    const afterDedupOutput = runData['Aggregate All Jobs']?.[0]?.output?.[0] || [];
    const afterDedup = afterDedupOutput.filter(i => !i.json?._empty).length;
    const dedupOutput = runData['Deduplicate vs Pipeline']?.[0]?.output?.[0] || [];
    const netNewPipelineRecords = dedupOutput.filter(i => !i.json?._empty).length;
    const pipelineRecordsCreated = runData['Create Pipeline Records']?.[0]?.output?.[0]?.length || 0;
    const errorInfo = executionData.data?.resultData?.error || executionData.error || null;
    const executionStatus = executionData.status || 'unknown';
    const feedsGeneratedCount = feedsGenerated;

    const valid = feedsGeneratedCount > 0 && executionStatus !== 'error';

    const perFeedItems = (runData['RSS Feed Read'] || []).map((iter, idx) => ({
      feedIndex: idx,
      items: (iter.output?.[0] || []).filter(i => !i.json?._empty).length,
      total: (iter.output?.[0] || []).length
    }));

    const perFeedAfterGeo = (runData['Parse & Filter Jobs'] || []).map((iter, idx) => ({
      feedIndex: idx,
      items: (iter.output?.[0] || []).filter(i => !i.json?._empty).length
    }));

    return [{
      json: {
        executionId: executionData.id,
        executionStatus,
        valid,
        error: errorInfo ? errorInfo.message || JSON.stringify(errorInfo) : null,
        errorDetails: errorInfo ? JSON.stringify({ message: errorInfo.message, lastNodeExecuted: executionData.lastNodeExecuted }) : null,
        startedAt: executionData.startedAt,
        stoppedAt: executionData.stoppedAt,
        runDurationSeconds: executionData.startedAt && executionData.stoppedAt
          ? (new Date(executionData.stoppedAt) - new Date(executionData.startedAt)) / 1000 : 0,
        feedsGenerated: feedsGeneratedCount,
        totalRssItems,
        perFeedItems: JSON.stringify(perFeedItems),
        afterGeoFilter,
        perFeedAfterGeo: JSON.stringify(perFeedAfterGeo),
        afterDedup,
        netNewPipelineRecords,
        pipelineRecordsCreated
      }
    }];
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| n8n node → n8n internal API | Internal read-only query of execution data |
| Airtable node → Airtable API | Outbound write to Baseline Metrics table |
| Error Trigger → measurement workflow | Error data from 01e crosses workflow boundary |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-01 | Information Disclosure | Airtable PAT in workflow JSON | mitigate | Reuse existing `airtableTokenApi` n8n credential — never inline PAT in JSON. Credential ID is a reference, not the token. |
| T-03-02 | Information Disclosure | n8n API key in workflow JSON | mitigate | Store as Header Auth n8n credential (not inline). User assigns post-import via credential selector. |
| T-03-04 | Spoofing | Error Trigger receives data from 01e | accept | Error data is execution metadata only (execution ID, error message). No write capability from Error path. |
| T-03-05 | Tampering | Execution data from n8n API | accept | Read-only query. Data originates from n8n's own PostgreSQL, produced by 01e workflow execution. No user-controllable injection path. |
| T-03-SC | Tampering | Package installs | mitigate | No packages installed — n8n built-in nodes only. |
</threat_model>
